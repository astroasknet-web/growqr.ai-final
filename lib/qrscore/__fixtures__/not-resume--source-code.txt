import os
import sys

def read_config(path):
    with open(path) as f:
        data = f.read()
    return data.split("\n")

class Handler:
    def __init__(self, name):
        self.name = name

    def run(self):
        for line in read_config(self.name):
            print(line)

if __name__ == "__main__":
    Handler(sys.argv[1]).run()
